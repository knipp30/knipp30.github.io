Configure Health Probes for VMs

Run strategy parameters, health probes, watchdog devices to monitor VM health.

Run Strategy Parameters

# spec.runStrategy controls VM lifecycle behavior

# RunStrategy values:
# Always:      VMI always present (new VMI created if original stops)
# RerunOnFailure: VMI re-created on failure, NOT on stop
# Manual:      User controls via virtctl start/stop/restart
# Halted:      No VMI present when VM created

# Run strategy transitions:
# Start command:
#   - Always → Always
#   - RerunOnFailure → RerunOnFailure
#   - Manual → Manual
#   - Halted → Always
# Stop command:
#   - Always → Halted
#   - RerunOnFailure → RerunOnFailure
#   - Manual → Manual
# Restart command:
#   - Always → Always
#   - RerunOnFailure → RerunOnFailure
#   - Manual → Manual

Configuring Run Strategy

# Configure VM run strategy via CLI
cat > vm-run-strategy.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  runStrategy: Always  # or RerunOnFailure, Manual, Halted
  template:
    spec:
      domain: {}
EOF

# Edit existing VM run strategy
oc edit vm my-vm -n my-namespace
# Update: spec.runStrategy: Always

# virtctl commands affect run strategy:
# virtctl start → VM will run based on strategy
# virtctl stop → VM halted (but strategy preserved)
# virtctl restart → VM restarted based on strategy

Health Probes

# Health probes monitor VM health and responsiveness
# Types:
# - Liveness: Determines if VM is running, restarts if probe fails
# - Readiness: Determines if VM is ready to accept traffic
# - Guest agent ping: Uses QEMU guest agent to check VM responsiveness

# Liveness probe (TCP)
cat > vm-liveness.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
spec:
  template:
    spec:
      domain:
        devices: {}
      readinessProbe:
        tcpSocket:
          port: 80
        initialDelaySeconds: 30
        periodSeconds: 10
        timeoutSeconds: 5
        successThreshold: 1
        failureThreshold: 3
      livenessProbe:
        tcpSocket:
          port: 22
        initialDelaySeconds: 30
        periodSeconds: 10
EOF

# Guest agent ping probe
cat > vm-guest-ping.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
spec:
  template:
    spec:
      domain:
        devices:
          guestAgent: {}
      guestAgent:
        enabled: true
      livenessProbe:
        guestAgentPing: {}
EOF

Watchdog Devices

# Watchdog monitors VM and node health
# Triggers actions when VM becomes unresponsive

# Configure watchdog device
cat > vm-watchdog.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
spec:
  template:
    spec:
      domain:
        devices:
          watchdogs:
          - name: my-watchdog
            i6300esb:
              action: reset
EOF

# Watchdog actions:
# reset:     Reboot the VM (default)
# powerOff:  Power down the VM
# shutdown:  Graceful shutdown
# pause:     Pause the VM

# Configure watchdog with node health check
cat > vm-watchdog-nhc.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
spec:
  template:
    spec:
      domain:
        devices:
          watchdogs:
          - name: nhc-watchdog
            i6300esb:
              action: reset
EOF

VM Health Monitoring

# Monitor VM health
oc get vm my-vm -n my-namespace -o jsonpath='{.status.conditions}'

# Check VM instance status
oc get vmi my-vm -n my-namespace -o jsonpath='{.status.phase}'

# VM conditions:
# - VMReady: VM is ready (running or stopped)
# - Paused: VM is paused
# - Ready: VMI is ready

# Monitoring via web console:
# Virtualization → Overview → Status card
# Shows overall health based on alerts and conditions
