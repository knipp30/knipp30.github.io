Connect VMs to External Networks

Multus CNI, NMstate operator, multihomed nodes and VMs, user-defined network connections.

Multus CNI Plug-in

# Multus CNI is a CNCF network plug-in that enables attaching multiple network interfaces to pods/VMs
# It provides secondary network interfaces alongside the primary OVN-Kubernetes network

# Multus is required for:
# - User-defined networks (UDN)
# - SR-IOV networking
# - Linux bridge networks
# - Dedicated live migration networks
# - External network access for VMs

# Verify Multus installation
oc get pods -n openshift-multus
oc get daemonset -n openshift-multus

# Multus NetworkAttachmentDefinition
apiVersion: k8s.cni.cncf.io/v1beta1
kind: NetworkAttachmentDefinition
metadata:
  name: my-multus-network
  namespace: my-namespace
spec:
  config: '{
    "cniVersion": "0.3.1",
    "type": "whereabouts",
    "range": "10.1.0.0/24",
    "gateway": "10.1.0.1"
  }'

NMstate Operator

# NMstate operator manages node network configuration
# Used for Linux bridge networks, SR-IOV, and dedicated live migration networks

# Install NMstate operator
oc apply -f - <<EOF
apiVersion: operators.coreos.com/v1alpha1
kind: Subscription
metadata:
  name: nmstate
  namespace: openshift-nmstate
spec:
  channel: stable
  name: kubevirt-nmstate
  source: redhat-operators
  sourceNamespace: openshift-marketplace
  installPlanApproval: Automatic
EOF

# Verify installation
oc get pods -n openshift-nmstate

# NMstate NetworkConfigurationPolicy (NNCP)
cat > nncp-linux-bridge.yaml <<EOF
apiVersion: nmstate.io/v1beta1
kind: NetworkConfigurationPolicy
metadata:
  name: linux-bridge-nncp
spec:
  nodeSelector:
    node-role.kubernetes.io/worker: ""
  desiredState:
    interfaces:
    - name: eth1
      type: ethernet
      state: up
EOF

oc apply -f nncp-linux-bridge.yaml

Configuring Linux Bridge Network

# Linux bridge networks connect VMs to external VLANs
# Most efficient method for connecting VMs to VLANs

# Step 1: Create NNCP (Network Configuration Policy)
cat > linux-bridge-nncp.yaml <<EOF
apiVersion: nmstate.io/v1beta1
kind: NetworkConfigurationPolicy
metadata:
  name: linux-bridge-policy
  namespace: openshift-nmstate
spec:
  nodeSelector:
    node-role.kubernetes.io/worker: ""
  desiredState:
    interfaces:
    - name: eth1
      type: ethernet
      state: up
    bridge:
      options-database:
        stp:
          mode: listening
        forward-delay: 15
        hello-time: 15
      interface:
        name: br-ext
        type: linux-bridge
        state: up
EOF

oc apply -f linux-bridge-nncp.yaml

# Step 2: Create NAD (Network Attachment Definition)
cat > linux-bridge-nad.yaml <<EOF
apiVersion: k8s.cni.cncf.io/v1beta1
kind: NetworkAttachmentDefinition
metadata:
  name: linux-bridge-network
  namespace: my-namespace
spec:
  config: '{
    "cniVersion": "0.3.1",
    "type": "macvtap",
    "master": "br-ext",
    "mode": "bridge",
    "ipam": {}
  }'
EOF

oc apply -f linux-bridge-nad.yaml

# Connect VM to Linux bridge network
cat > vm-linux-bridge.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
spec:
  template:
    spec:
      domain:
        devices:
          interfaces:
          - name: br-net
            bridge: {}
            macvtap: {}
            multus:
              network: my-namespace/linux-bridge-network
      networks:
      - name: br-net
        multus:
          networkName: my-namespace/linux-bridge-network
EOF

Multihomed Nodes and VMs

# Multihomed nodes: Nodes with multiple network interfaces
# Multihomed VMs: VMs with multiple network interfaces

# Configure dedicated network for live migration
cat > live-migration-network.yaml <<EOF
apiVersion: k8s.cni.cncf.io/v1beta1
kind: NetworkAttachmentDefinition
metadata:
  name: live-migration
  namespace: openshift-cnv
spec:
  config: '{
    "cniVersion": "0.3.1",
    "type": "whereabouts",
    "range": "10.56.0.0/24",
    "gateway": "10.56.0.1"
  }'
EOF

# Attach multiple interfaces to VM
cat > vm-multihomed.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
spec:
  template:
    spec:
      domain:
        devices:
          interfaces:
          - name: primary
            masquerade: {}
          - name: secondary
            bridge: {}
            macvtap: {}
            multus:
              network: my-namespace/external-network
          - name: live-mig
            multus:
              network: openshift-cnv/live-migration
      networks:
      - name: primary
        pod: {}
      - name: secondary
        multus:
          networkName: my-namespace/external-network
      - name: live-mig
        multus:
          networkName: openshift-cnv/live-migration
EOF

Configuring SR-IOV Network

# SR-IOV provides high-performance networking via hardware virtualization
# Not supported on s390x architecture

# Install SR-IOV Network Operator
cat > sriov-operator.yaml <<EOF
apiVersion: operators.coreos.com/v1alpha1
kind: Subscription
metadata:
  name: sriov-network-operator
  namespace: openshift-sriov-network-operator
spec:
  channel: stable
  name: sriov-network-operator
  source: redhat-operators
  sourceNamespace: openshift-marketplace
EOF

# SR-IOV NetworkConfig
cat > sriov-network.yaml <<EOF
apiVersion: sriov.network.k8s.io/v1
kind: SriovNetwork
metadata:
  name: sriov-net
  namespace: my-namespace
spec:
  networkNamespace: my-namespace
  resourceName: vendor.name.sriov-net
  ipam: {}
  preferredInterface: sriov
EOF

# SR-IOV Network Attachment Definition
cat > sriov-nad.yaml <<EOF
apiVersion: k8s.cni.cncf.io/v1beta1
kind: NetworkAttachmentDefinition
metadata:
  name: sriov-network
  namespace: my-namespace
spec:
  config: '{
    "cniVersion": "0.3.1",
    "type": "sriov"
  }'
EOF

Dedicated Secondary Network for Live Migration

# Dedicated Multus network for live migration is highly recommended
# Minimizes effects of network saturation on tenant workloads

# Configure for live migration via NMstate
cat > nm-migration-nncp.yaml <<EOF
apiVersion: nmstate.io/v1beta1
kind: NetworkConfigurationPolicy
metadata:
  name: migration-network-policy
spec:
  nodeSelector:
    node-role.kubernetes.io/worker: ""
  desiredState:
    interfaces:
    - name: bond0
      type: bond
      state: up
EOF

# Configure HCO to use dedicated network
cat > hc-migration-net.yaml <<EOF
apiVersion: hco.kubevirt.io/v1beta1
kind: HyperConverged
metadata:
  name: kubevirt-hyperconverged
  namespace: openshift-cnv
spec:
  configuration:
    migrationNetwork: migration-network
EOF
