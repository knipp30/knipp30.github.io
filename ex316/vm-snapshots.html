Manage Virtual Machine Snapshots

Create and manage VM snapshots, restore from snapshots.

About VM Snapshots

# VM Snapshot represents state and data at a specific point in time
# Supports:
# - Offline snapshots (powered off VM)
# - Online snapshots (running VM)

# Snapshot requirements:
# - Disks must be DataVolumes or PVCs
# - Disks must belong to storage class supporting CSI volume snapshots
# - Disks must be bound to PV and populated with datasource
# - vTPM device: cloning from snapshot is NOT supported

# Snapshot stores:
# - Copy of each CSI volume attached to VM
# - Copy of VM specification and metadata

# IMPORTANT: Snapshots are immutable - cannot be changed after creation

VM Snapshot CRDs

# Three CRDs manage snapshots:
# 1. VirtualMachineSnapshot: User request to create snapshot
# 2. VirtualMachineSnapshotContent: Provisioned snapshot resource
# 3. VirtualMachineRestore: User request to restore from snapshot

# Controller binds VirtualMachineSnapshotContent to VirtualMachineSnapshot
# One-to-one mapping between snapshot and snapshot content

Creating Snapshots

# Via Web Console:
# 1. Navigate to Virtualization → VirtualMachines
# 2. Select VM → Snapshots tab
# 3. Click Take Snapshot
# 4. Enter snapshot name
# 5. Expand "Disks included in this Snapshot" to verify
# 6. If disks cannot be included, check "I am aware of this warning"
# 7. Click Save

# Via CLI:
cat > snapshot.yaml <<EOF
apiVersion: snapshot.kubevirt.io/v1beta1
kind: VirtualMachineSnapshot
metadata:
  name: my-snapshot
  namespace: my-namespace
spec:
  source:
    apiGroup: kubevirt.io
    kind: VirtualMachine
    name: my-vm
EOF

oc create -f snapshot.yaml

# Monitor snapshot creation
oc wait virtualmachinesnapshot my-snapshot --for condition=Ready -n my-namespace

# Check status: InProgress → Succeeded/Failed

Online Snapshots

# Online snapshot requirements:
# - QEMU guest agent must be installed and running
# - Agent freezes/thaws filesystem for consistency
# - Without agent: best-effort snapshot only

# Default online snapshot timeout: 5 minutes
# Change with FailureDeadline attribute:
cat > snapshot-timeout.yaml <<EOF
apiVersion: snapshot.kubevirt.io/v1beta1
kind: VirtualMachineSnapshot
metadata:
  name: my-snapshot
  namespace: my-namespace
spec:
  source:
    apiGroup: kubevirt.io
    kind: VirtualMachine
    name: my-vm
  failureDeadline: 10m
EOF

# Snapshot indications:
# - Online: VM was running during snapshot
# - GuestAgent: QEMU guest agent was running (quiesced)
# - NoGuestAgent: QEMU guest agent not running

# Check indications:
oc describe vmsnapshot my-snapshot -n my-namespace | grep -A 5 indications

# Enable QEMU guest agent:
# On Linux: yum install qemu-guest-agent && systemctl enable --now qemu-guest-agent
# On Windows: Install VirtIO drivers, run qemu-ga-x86_64.msi

Restoring from Snapshots

# Via Web Console:
# 1. Navigate to Virtualization → VirtualMachines
# 2. Select VM → Snapshots tab
# 3. If running, Options → Stop
# 4. Select snapshot → Options → Restore VirtualMachine from snapshot
# 5. Click Restore

# Via CLI - Restore existing VM:
cat > restore.yaml <<EOF
apiVersion: snapshot.kubevirt.io/v1beta1
kind: VirtualMachineRestore
metadata:
  name: my-restore
  namespace: my-namespace
spec:
  source:
    apiGroup: snapshot.kubevirt.io
    kind: VirtualMachineSnapshotContent
    name: my-snapshot-content
  virtualMachineName: my-vm
EOF

oc create -f restore.yaml

# Monitor restore
oc wait virtualmachinerestore my-restore --for condition=Ready -n my-namespace

# Restore to new VM (create from snapshot):
cat > new-vm-snapshot.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm-cloned
  namespace: my-namespace
spec:
  runStrategy: Always
  template:
    spec:
      domain: {}
EOF

# Verify restore
oc describe virtualmachinerestore my-restore -n my-namespace

Managing Snapshots

# List all snapshots
oc get virtualmachinesnapshots -n my-namespace

# List snapshot contents
oc get virtualmachinesnapshotcontent -n my-namespace

# Delete snapshot
oc delete virtualmachinesnapshot my-snapshot -n my-namespace

# Check snapshot status
oc describe vmsnapshot my-snapshot -n my-namespace

# Expected conditions:
# Progressing: status="False" when complete
# Ready: status="True" when ready to use
# readyToUse: true
