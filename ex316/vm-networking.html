Configure Kubernetes Networking for VMs

Kubernetes SDN, network policies, external access, user-defined networks, ClusterIP services.

Networking Overview

# Key networking concepts
# - Multus CNI: Secondary network plug-in for VMs
# - Default pod network: OVN-Kubernetes primary SDN
# - User-defined networks (UDN): Primary or secondary networks for VMs
# - IPv6: Single-stack IPv6 NOT supported (dual-stack supported)
# - SR-IOV: For high-performance networking
# - Linux bridge: For external access via VLAN

# Layer2 topology recommended on public clouds for UDNs
# IPAM (IP address management) in NAD for VMs is NOT supported

Connecting VMs to Default Pod Network

# Masquerade mode (default): VM has outbound NAT, can access external network
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  template:
    spec:
      domain:
        devices:
          interfaces:
          - name: default
            masquerade: {}
            networkName: default
      networks:
      - name: default
        pod: {}

# IPv4 with dual-stack support
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-dualstack-vm
spec:
  template:
    spec:
      domain:
        devices:
          interfaces:
          - name: default
            masquerade: {}
        networkInterface: 192.168.1.1
      networks:
      - name: default
        pod: {}
      # Cloud-init for IPv6
      initContainers:
      - name: ipv6-setup
        image: redhat/ovirt-guest-agent

Configuring Primary User-Defined Network

# Create namespace for UDNs (web console)
# 1. Navigate to Administration → Namespaces
# 2. Create project

# Create primary UDN by CLI
cat > nad-primary.yaml <<EOF
apiVersion: k8s.cni.cncf.io/v1beta1
kind: NetworkAttachmentDefinition
metadata:
  name: primary-network
  namespace: my-namespace
spec:
  config: '{
    "cniVersion": "0.3.1",
    "type": "macvtap",
    "master": "eth0",
    "mode": "bridge",
    "ipam": {}
  }'
EOF

oc apply -f nad-primary.yaml

# Attach VM to primary UDN
cat > vm-primary-udn.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm-udn
spec:
  template:
    spec:
      domain:
        devices:
          interfaces:
          - name: udn
            primaryIP: 10.0.0.1
            multus:
              network: my-namespace/primary-network
      networks:
      - name: udn
        multus:
          networkName: my-namespace/primary-network
EOF

Exposing VMs with Services (ClusterIP, NodePort, LoadBalancer)

# ClusterIP service (default, internal only)
cat > vm-clusterip-service.yaml <<EOF
apiVersion: v1
kind: Service
metadata:
  name: my-vm-service
  namespace: my-namespace
  annotations:
    kubevirt.io/managedVirtualNetwork: default
spec:
  type: ClusterIP
  selector:
    kubevirt.io/managedVirtualNetwork: default
  ports:
  - port: 80
    targetPort: 80
    protocol: TCP
EOF

oc apply -f vm-clusterip-service.yaml

# NodePort service (external access via node IP)
cat > vm-nodeport-service.yaml <<EOF
apiVersion: v1
kind: Service
metadata:
  name: my-vm-nodeport
  namespace: my-namespace
spec:
  type: NodePort
  selector:
    kubevirt.io/managedVirtualNetwork: default
  ports:
  - port: 80
    targetPort: 80
    nodePort: 30080
    protocol: TCP
EOF

# LoadBalancer service (cloud provider external LB)
cat > vm-lb-service.yaml <<EOF
apiVersion: v1
kind: Service
metadata:
  name: my-vm-lb
  namespace: my-namespace
spec:
  type: LoadBalancer
  selector:
    kubevirt.io/managedVirtualNetwork: default
  ports:
  - port: 80
    targetPort: 80
    protocol: TCP
EOF

# Access VM service
oc get svc my-vm-service -n my-namespace
oc get endpoints -n my-namespace

# Dual-stack services
cat > vm-dualstack-service.yaml <<EOF
apiVersion: v1
kind: Service
metadata:
  name: my-vm-dualstack
  namespace: my-namespace
spec:
  ipFamilies:
  - IPv4
  - IPv6
  ipFamilyPolicy: PreferDualStack
  type: ClusterIP
  selector:
    kubevirt.io/managedVirtualNetwork: default
  ports:
  - port: 80
    protocol: TCP
EOF

Headless Service (Internal FQDN)

# Create headless service for internal DNS resolution
cat > headless-svc.yaml <<EOF
apiVersion: v1
kind: Service
metadata:
  name: my-vm-headless
  namespace: my-namespace
spec:
  # clusterIP is implicitly None for headless services
  selector:
    kubevirt.io/name: my-vm
EOF

# Map VM to headless service
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
spec:
  template:
    spec:
      metadata:
        labels:
          kubevirt.io/name: my-vm
      hostname: my-vm-host
      subdomain: my-vm-headless
      # DNS domain: my-vm-host.my-vm-headless.my-namespace.svc.cluster.local

Network Policies for VMs

# Apply Kubernetes NetworkPolicy to restrict VM traffic
cat > network-policy.yaml <<EOF
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: vm-network-policy
  namespace: my-namespace
spec:
  podSelector:
    matchLabels:
      kubevirt.io/name: my-vm
  policyTypes:
  - Ingress
  - Egress
  ingress:
  - from:
    - namespaceSelector:
        matchLabels:
          project: trusted
    ports:
    - protocol: TCP
      port: 80
  egress:
  - to:
    - podSelector:
        matchLabels:
          role: database
    ports:
    - protocol: TCP
      port: 5432
EOF

oc apply -f network-policy.yaml

# Allow all ingress
cat > allow-all-ingress.yaml <<EOF
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: allow-all-ingress
  namespace: my-namespace
spec:
  podSelector: {}
  ingress:
  - {}
  policyTypes:
  - Ingress
EOF

oc apply -f allow-all-ingress.yaml

External Access for VMs

# Via NodePort service
oc expose vm my-vm --port=80 --name=my-vm-svc --type=NodePort

# Via LoadBalancer service (cloud platforms)
oc expose vm my-vm --port=80 --name=my-vm-svc --type=LoadBalancer

# Via route (for web services)
# Create Route for the Service
cat > vm-route.yaml <<EOF
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: my-vm-route
  namespace: my-namespace
spec:
  host: my-vm.example.com
  to:
    kind: Service
    name: my-vm-svc
  port:
    targetPort: 80
EOF

# Expose additional routes for cdi-uploadproxy
oc create route reencrypt cdi-uploadproxy \
  --service=cdi-uploadproxy \
  --hostname=upload.example.com \
  -n openshift-cdi
