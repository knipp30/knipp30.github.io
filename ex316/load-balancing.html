Manage VM Load Balancing

Load balancing with Kubernetes networking, NodePort services, customized routes.

VM Load Balancing with Services

# Expose VMs using Kubernetes services for load balancing

# NodePort service (external via node IP:port)
cat > vm-nodeport-service.yaml <<EOF
apiVersion: v1
kind: Service
metadata:
  name: my-vm-nodeport
  namespace: my-namespace
  annotations:
    kubevirt.io/managedVirtualNetwork: default
spec:
  type: NodePort
  selector:
    kubevirt.io/name: my-vm
  ports:
  - name: http
    port: 80
    targetPort: 80
    nodePort: 30080
    protocol: TCP
EOF

oc apply -f vm-nodeport-service.yaml

# LoadBalancer service (cloud provider managed)
cat > vm-lb-service.yaml <<EOF
apiVersion: v1
kind: Service
metadata:
  name: my-vm-lb
  namespace: my-namespace
  annotations:
    kubevirt.io/managedVirtualNetwork: default
spec:
  type: LoadBalancer
  selector:
    kubevirt.io/name: my-vm
  ports:
  - name: http
    port: 80
    targetPort: 80
    protocol: TCP
EOF

# Access external IP
oc get svc my-vm-lb -n my-namespace
# Get the EXTERNAL-IP address

NodePort Service Configuration

# Create NodePort service via CLI
oc expose vm my-vm --port=80 --name=my-vm-svc --type=NodePort -n my-namespace

# Verify service
oc get svc my-vm-svc -n my-namespace
oc get endpoints -n my-namespace

# Access via any node IP on NodePort
# http://<node-ip>:<node-port>

# Multiple ports
cat > multiport-svc.yaml <<EOF
apiVersion: v1
kind: Service
metadata:
  name: my-vm-multiport
  namespace: my-namespace
spec:
  type: NodePort
  selector:
    kubevirt.io/name: my-vm
  ports:
  - name: http
    port: 80
    targetPort: 80
    nodePort: 30080
    protocol: TCP
  - name: https
    port: 443
    targetPort: 443
    nodePort: 30443
    protocol: TCP
EOF

Customized Routes

# Create custom routes to expose VM services
cat > vm-route.yaml <<EOF
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: my-vm-route
  namespace: my-namespace
spec:
  host: my-vm.example.com
  to:
    kind: Service
    name: my-vm-svc
    weight: 100
  port:
    targetPort: 80
  tls:
    termination: reencrypt
    key: |
      -----BEGIN PRIVATE KEY-----
      ...
    certificate: |
      -----BEGIN CERTIFICATE-----
      ...
EOF

oc apply -f vm-route.yaml

# Custom routes for cdi-uploadproxy
oc create route reencrypt cdi-uploadproxy \
  --service=cdi-uploadproxy \
  --hostname=upload.example.com \
  -n openshift-cdi

# Route with edge termination
cat > edge-route.yaml <<EOF
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: my-vm-edge-route
  namespace: my-namespace
spec:
  host: my-vm.example.com
  to:
    kind: Service
    name: my-vm-svc
  tls:
    termination: edge
EOF

Enabling Load Balancer Service Creation

# Enable load balancer service creation via web console
# Virtualization → Overview → Settings → Cluster → Enable load balancer

# Via Web Console (add services):
# 1. Navigate to Virtualization → VirtualMachines
# 2. Select a VM → Services tab
# 3. Click "Add Service"
# 4. Select type: LoadBalancer, NodePort, or ClusterIP
# 5. Configure ports (port, targetPort, protocol)
# 6. For LoadBalancer: select load balancer provider
# 7. Click Create

# To add custom routes:
# 1. Navigate to Networking → Routes
# 2. Click "Create Route"
# 3. Select namespace and service
# 4. Configure hostname, TLS termination
# 5. Click Create

# Or via CLI
cat > hc-enable-lb.yaml <<EOF
apiVersion: hco.kubevirt.io/v1beta1
kind: HyperConverged
metadata:
  name: kubevirt-hyperconverged
  namespace: openshift-cnv
spec:
  configuration:
    loadBalancer:
      enabled: true
EOF

oc apply -f hc-enable-lb.yaml
