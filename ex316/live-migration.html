Perform Virtual Machine Live Migrations

Migration limitations, node affinity, start/monitor/cancel VM migrations.

Live Migration Overview

# Live migration requirements:
# - Shared storage with ReadWriteMany (RWX) access mode
# - Sufficient RAM and network bandwidth
# - Dedicated Multus network for live migration (highly recommended)
# - Nodes must support the VM's CPU host model (if configured)
# - Default: max 5 parallel migrations in cluster

# Live migration NOT supported with:
# - RWO (ReadWriteOnce) storage volume
# - GPU passthrough (PCI devices)
# - SR-IOV devices
# - TPM devices
# Set evictionStrategy: None for non-migratable VMs

Live Migration Limitations

# Cannot live migrate with:
# - ReadWriteOnce (RWO) access mode storage
# - Passthrough features (GPUs)
# - Host model CPU on nodes with different CPUs
# - Single-node OpenShift
# - Non-migratable VMs

# Non-migratable VMs should set evictionStrategy:
# - LiveMigrateIfPossible: Does not block upgrade
# - None: Shuts down during node maintenance

# Memory overhead calculation for live migration capacity:
# Required spare memory = Max parallel drains × Highest total VM memory request per node
# Default parallel migrations: 5

Node Affinity for VMs

# Configure required node affinity rule
cat > vm-affinity.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-affinity-vm
  namespace: my-namespace
spec:
  template:
    spec:
      affinity:
        nodeAffinity:
          requiredDuringSchedulingIgnoredDuringExecution:
            nodeSelectorTerms:
            - matchExpressions:
              - key: node-role.kubernetes.io/compute
                operator: In
                values:
                - "true"
              - key: topology.kubernetes.io/zone
                operator: In
                values:
                - zone-a
EOF

oc apply -f vm-affinity.yaml

# Node selector (simpler)
cat > vm-node-selector.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-node-selector-vm
  namespace: my-namespace
spec:
  template:
    spec:
      nodeSelector:
        node-role.kubernetes.io/compute: "true"
EOF

oc apply -f vm-node-selector.yaml

Starting Migrations

# Start a live migration via CLI
virtctl migrate <vm_name> -n <namespace>

# Start migration for VM with migration config
cat > migrate-config.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
spec:
  template:
    spec:
      migrationStrategy: Migrate
      terminationGracePeriodSeconds: 180
EOF

# Set eviction strategy for migratable VMs
cat > vm-eviction.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  template:
    spec:
      evictionStrategy: LiveMigrate
EOF

oc apply -f vm-eviction.yaml

# Restart VM to apply eviction strategy changes
virtctl restart my-vm -n my-namespace

Monitoring Migrations

# Check migration status
oc get vmim -n my-namespace

# VMIM phases:
# - New: Migration request received
# - Pending: Migration waiting to start
# - Preparing: Pre-migration checks running
# - Running: Migration in progress
# - Succeeded: Migration completed successfully
# - Failed: Migration failed
# - Canceled: Migration was canceled

# Detailed migration status
oc get vmim <migration_name> -n my-namespace -o yaml

# Monitor migration progress
oc describe vmim <migration_name> -n my-namespace

# Migration events
oc get events -n my-namespace --field-selector reason=MigrationTargetFree

# Check VM status during migration
oc get vm my-vm -n my-namespace
# VMI phase should show Migrated during migration

# Migration configuration (cluster-wide)
oc edit hyperconverged kubevirt-hyperconverged -n openshift-cnv
# Set migration parameters:
#   spec.configuration.migrationConfiguration:

Canceling Migrations

# Cancel a running migration
oc patch vmim <migration_name> -n my-namespace --type merge \
  -p '{"spec":{"abortRequested":true}}'

# Verify cancellation
oc get vmim <migration_name> -n my-namespace
# Phase should change to Canceled

# If migration is stuck, check for blockers
oc get vmim -n my-namespace -o jsonpath='{range .items[*]}{.metadata.name}{"\t"}{.status.phase}{"\n"}{end}'

Cluster Eviction Strategy

# Configure cluster-wide eviction strategy
oc edit hyperconverged kubevirt-hyperconverged -n openshift-cnv

# Cluster eviction strategies:
# LiveMigrate:     Prioritizes workload continuity (default multi-node)
#                  - Interrupts workflow: No
#                  - Blocks upgrades: Yes
# LiveMigrateIfPossible: Prioritizes upgrades (does not block)
#                        - Interrupts workflow: Yes
#                        - Blocks upgrades: No
# None:            Shuts down VMs
#                  - Default for single-node OpenShift

apiVersion: hco.kubevirt.io/v1beta1
kind: HyperConverged
metadata:
  name: kubevirt-hyperconverged
  namespace: openshift-cnv
spec:
  evictionStrategy: LiveMigrate
  configuration:
    migrationConfiguration:
      batchEvictionSize: 5
      batchEvictionInterval: 1m
      parallelOutboundMigrations: 5
      nodeDrainTaintKeys:
      - kubevirt.io/drain
