Configure Kubernetes Storage for VMs

Persistent storage, disk attach/detach, external storage via Multus, data volumes, CDI.

Storage Configuration Overview

# Mandatory tasks:
# - Configure a default storage class
# - Configure storage profiles for recognized providers

# Optional tasks:
# - Reserve additional PVC space for file system overhead
# - Configure local storage via HPP
# - Configure RBAC for cloning data volumes between namespaces

# Storage access modes:
# - ReadWriteOnce (RWO): Single node (NOT supported for live migration)
# - ReadWriteMany (RWX): Multiple nodes (REQUIRED for live migration)
# - ReadOnlyMany: Multiple read-only nodes

# Volume modes:
# - Block: Direct disk access (recommended, better performance)
# - Filesystem: File system layer + disk image (more layers, slower)

# IMPORTANT: RWX + Block is recommended for best performance and live migration

# Mark storage class as default for virtualization
# Set annotation: storageclass.kubevirt.io/is-default-virt-class: "true"

Configuring Storage Profiles

# CDI recognizes storage providers automatically
# For unrecognized providers, configure storage profiles manually

# Check storage profiles
oc get storageprofile

# Edit storage profile for unrecognized provider
oc edit storageprofile <profile-name>

# Customizing storage profile
cat > storage-profile-edit.yaml <<EOF
apiVersion: cdi.kubevirt.io/v1beta1
kind: StorageProfile
metadata:
  name: my-storage-class
spec:
  pvc:
    accessModes:
    - ReadWriteMany
    volumeMode: Block
EOF

# For ODF with Ceph RBD (recommended for OCP Virtualization)
# Use 'ocs-storagecluster-ceph-rbd' storage class with VolumeMode: Block
# RBD block mode is more efficient than CephFS or RBD filesystem-mode PVCs

# WARNING: If disk has more than one VolumeSnapshotClass, 
# you must specify one before creating snapshots

# Custom storage profile parameters
cat > custom-storage.yaml <<EOF
apiVersion: cdi.kubevirt.io/v1beta1
kind: StorageProfile
metadata:
  name: my-custom-class
spec:
  pvc:
    accessModes: ["ReadWriteMany"]
    volumeMode: Block
    storage:
      accessModes: ["ReadWriteMany"]
      resources:
        requests:
          storage: 10Gi
EOF

Containerized Data Importer (CDI)

# CDI manages VM disk imports, uploads, and cloning

# Override CDI resource requests for namespaces with quotas
cat > cdi-override.yaml <<EOF
apiVersion: cdi.kubevirt.io/v1beta1
kind: CDI
metadata:
  name: cdi
spec:
  config:
    featureGates: {}
    resourceRequests:
      importer:
        cpu: "1"
        memory: "1Gi"
EOF

# Configure CDI scratch space
# CDI provisions scratch PVC equal to size of destination DV during import/upload

# Enable preallocation for data volumes (improves write performance)
cat > dv-prealloc.yaml <<EOF
apiVersion: cdi.kubevirt.io/v1beta1
kind: DataVolume
metadata:
  name: preallocated-dv
  namespace: my-namespace
  annotations:
    cdi.kubevirt.io/storage.preallocation: "true"
spec:
  source:
    http:
      url: https://example.com/image.qcow2
  storage:
    resources:
      requests:
        storage: 10Gi
EOF

# Manage data volume annotations
cat > dv-annotations.yaml <<EOF
apiVersion: cdi.kubevirt.io/v1beta1
kind: DataVolume
metadata:
  name: annotated-dv
  namespace: my-namespace
  annotations:
    cdi.kubevirt.io/storage.overwriteable: "true"
    cdi.kubevirt.io/storage.forceBind: "true"
spec:
  storage:
    resources:
      requests:
        storage: 10Gi
EOF

Data Volumes (DVs)

# DataVolume objects manage VM disk storage

# Create DataVolume from URL
cat > dv-http.yaml <<EOF
apiVersion: cdi.kubevirt.io/v1beta1
kind: DataVolume
metadata:
  name: my-image-dv
  namespace: my-namespace
spec:
  source:
    http:
      url: https://example.com/rhel9.qcow2
  storage:
    resources:
      requests:
        storage: 10Gi
    storageClassName: ocs-storagecluster-ceph-rbd
EOF

# Create DataVolume for upload
cat > dv-upload.yaml <<EOF
apiVersion: cdi.kubevirt.io/v1beta1
kind: DataVolume
metadata:
  name: my-upload-dv
  namespace: my-namespace
spec:
  source:
    blank: {}
  storage:
    resources:
      requests:
        storage: 20Gi
    storageClassName: ocs-storagecluster-ceph-rbd
EOF

# Monitor DataVolume status
oc get dv my-image-dv -n my-namespace
# Expected phases: Empty, Create, WaitForFirstConsumer, Bound, Succeeded

# Upload image to DataVolume
virtctl upload disk \
  --data-volume=my-upload-dv \
  --size=20Gi \
  --image-path=./my-image.qcow2 \
  -n my-namespace

Attaching Disks to VMs

# Attach additional disk to running VM
cat > vm-with-disk.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm-with-disk
spec:
  template:
    spec:
      domain:
        devices:
          disks:
          - name: rootdisk
            disk:
              bus: virtio
          - name: datadisk
            disk:
              bus: virtio
          interfaces:
          - name: default
            masquerade: {}
      volumes:
      - name: rootdisk
        persistentVolumeClaim:
          claimName: my-vm-pvc
      - name: datadisk
        persistentVolumeClaim:
          claimName: additional-disk
EOF

# Hot-plug disk to running VM
oc patch vm my-vm -n my-namespace --type merge -p '{"spec":{"template":{"spec":{"volumes":[{"name":"hotplug-disk","persistentVolumeClaim":{"claimName":"hotplug-pvc"}}]}}}}'

# Hot-plug device to running VM
oc patch vm my-vm -n my-namespace --type merge -p '{"spec":{"template":{"spec":{"domain":{"devices":{"gpus":[{"name":"gpu1","deviceName":"nvidia.com/GPU"}]}}}}}}'

# Create and attach disk
cat > add-disk.yaml <<EOF
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: extra-disk
  namespace: my-namespace
spec:
  accessModes:
  - ReadWriteOnce
  resources:
    requests:
      storage: 5Gi
  storageClassName: ocs-storagecluster-ceph-rbd
EOF

# Attach via virtctl
virtctl add-volume disk-name <vm_name> -n <namespace>

Detaching Disks from VMs

# Hot-unplug volume from running VM
oc patch vm my-vm -n my-namespace --type merge -p '{"spec":{"template":{"spec":{"volumes":[{"name":"disk-to-remove","persistentVolumeClaim":{"claimName":"my-pvc"}}],"volumeMounts":[{"name":"disk-to-remove","mountPath":"/mnt/data"}]}}}}'

# Hot-unplug device from running VM
virtctl remove-device device-name <vm_name> -n <namespace>

# Remove volume from VM
virtctl remove-volume volume-name <vm_name> -n <namespace>

# Delete PVC after detach (ensure no active references)
oc delete pvc extra-disk -n my-namespace

External Storage via Multus

# Connect VMs to external storage using Multus

# Storage class for external storage
cat > external-storage-class.yaml <<EOF
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: external-storage
  annotations:
    storageclass.kubevirt.io/is-default-virt-class: "true"
provisioner: external.storage.provisioner
parameters:
  storageType: external
  storageBackend: nfs
  server: "192.168.1.100"
  share: "/exports/vm-storage"
EOF

# PVC for external storage
cat > external-pvc.yaml <<EOF
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: external-vm-disk
  namespace: my-namespace
spec:
  accessModes:
  - ReadWriteMany
  resources:
    requests:
      storage: 10Gi
  storageClassName: external-storage
EOF

Local Storage with Hostpath Provisioner

# HPP is automatically installed with OpenShift Virtualization
# Configures local storage for VMs on worker nodes

# Create HostPathProvisioner
cat > hpp.yaml <<EOF
apiVersion: hostpathprovisioner.kubevirt.io/v1beta1
kind: HostPathProvisioner
metadata:
  name: hostpath-provisioner
  namespace: openshift-cnv
spec:
  unprivileged: true
  safeToEvict: true
  storage:
    images:
    - name: default-storage
      pathTemplate: "/var/run/kubevirt-storage"
      capacity: 100Gi
EOF

oc apply -f hpp.yaml

# Storage class for HPP
cat > hpp-sc.yaml <<EOF
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: hostpath-provisioner
  annotations:
    storageclass.kubevirt.io/is-default-virt-class: "true"
provisioner: hostpathprovisioner.storage.kubevirt.io
EOF

Boot Source Updates

# Manage automatic boot source updates

# Via Web Console:
# 1. Navigate to Virtualization → Storage → DataVolumes
# 2. To attach disk to VM:
#    a. Select VM → Disks tab
#    b. Click "Add Disk"
#    c. Select storage (PVC or DataVolume)
#    d. Set mount path and access mode
#    e. Click Add
# 3. To detach disk:
#    a. Select VM → Disks tab
#    b. Click "Remove" next to disk
#    c. Confirm removal
# 4. To configure storage profiles:
#    a. Navigate to Virtualization → Settings → Cluster
#    b. Scroll to "Storage" section
#    c. Edit storage profile settings
#    d. Click Save
# 5. To upload images:
#    a. Navigate to Virtualization → Catalog
#    b. Select blank template
#    c. Under Storage, choose "Upload"
#    d. Select image file, set size
#    e. Click Create
# CDI imports, polls, and updates images for cloning

# Enable auto-update for custom boot sources
cat > boot-source-ns.yaml <<EOF
apiVersion: v1
kind: Namespace
metadata:
  name: custom-boot-images
  labels:
    boot-source-updates: "true"
EOF
