Basic System Administration

Start/stop services, configure services at boot time, install software packages.

Managing Services with systemd

# Start a service
systemctl start <service_name>

# Stop a service
systemctl stop <service_name>

# Restart a service
systemctl restart <service_name>

# Reload service configuration (no downtime)
systemctl reload <service_name>

# Check service status
systemctl status <service_name>

# View service logs
journalctl -u <service_name> -f

# Common services for VMs:
systemctl start sshd
systemctl stop sshd
systemctl restart network

# Check if service is enabled
systemctl is-enabled <service_name>

Enabling Services at Boot

# Enable service to start at boot
systemctl enable <service_name>

# Enable and start immediately
systemctl enable --now <service_name>

# Disable service at boot
systemctl disable <service_name>

# List all enabled services
systemctl list-unit-files --type=service --state=enabled

# Common services to enable at boot:
systemctl enable --now sshd
systemctl enable --now firewalld
systemctl enable --now chronyd

# Mask service (prevent from ever starting)
systemctl mask <service_name>

# Unmask service
systemctl unmask <service_name>

Installing Software Packages

# DNF (RHEL 8+/Fedora)
# Install package
dnf install <package_name>

# Install multiple packages
dnf install vim net-tools curl

# Remove package
dnf remove <package_name>

# Search for package
dnf search <keyword>

# List installed packages
dnf list installed

# Update all packages
dnf update

# Update specific package
dnf update <package_name>

# List available packages
dnf list available

# Install from local RPM
dnf install ./package.rpm

# YUM (RHEL 7)
yum install <package_name>
yum remove &<package_name>
yum update
yum list installed

# RPM (low-level)
rpm -ivh <package_name>.rpm
rpm -Uvh <package_name>.rpm
rpm -e <package_name>
rpm -qa | grep <package_name>

Managing Software Repositories

# List configured repositories
dnf repolist
dnf repolist all

# Enable repository
dnf config-manager --set-enabled <repo_name>

# Disable repository
dnf config-manager --set-disabled <repo_name>

# Configure subscription manager
subscription-manager register
subscription-manager attach --auto
subscription-manager repos --enable <repo_id>
subscription-manager repos --disable <repo_id>

# Common repositories:
subscription-manager repos --enable=rhel-9-for-x86_64-baseos-rpms
subscription-manager repos --enable=rhel-9-for-x86_64-appstream-rpms

# Add custom repository
cat > /etc/yum.repos.d/custom.repo <<EOF
[custom-repo]
name=Custom Repository
baseurl=https://example.com/repo/rhel/9/x86_64/
enabled=1
gpgcheck=0
EOF

Service Management for VMs

# Manage services inside VMs via SSH
virtctl ssh root@<vm_name> -n my-namespace <<SCRIPT
systemctl start httpd
systemctl enable --now httpd
systemctl status httpd
SCRIPT

# Manage services inside VMs via console
virtctl console <vm_name> -n my-namespace

# Install software inside VM via cloud-init
# In cloud-init user data:
# #cloud-config
# packages:
#   - httpd
#   - net-tools
# runcmd:
#   - systemctl enable --now httpd

Additional System Administration Tasks

# Check disk usage
df -h
du -sh /var/log

# Check memory
free -m

# Check CPU
lscpu

# Network configuration
ip addr show
ip route show
nmcli connection show

# Firewall configuration
firewall-cmd --list-all
firewall-cmd --add-service=http --permanent
firewall-cmd --reload

# Time synchronization
timedatectl
chronyc tracking

# User management
useradd -m <username>
usermod -aG wheel <username>
passwd <username>
visudo
