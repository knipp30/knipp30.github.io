Prepare VMs for Node Failure

Eviction strategies, node selectors, affinity, anti-affinity, tolerations, taints, watchdog monitoring.

Eviction Strategies

# Eviction strategies determine VM behavior during node loss/maintenance

# Eviction strategies:
# LiveMigrate:      VM live migrated during node drain (default multi-node)
#                    - VMs with non-migratable config block drains
#                    - Set non-migratable VMs to LiveMigrateIfPossible
# LiveMigrateIfPossible: Upgrade prioritized, VMs not blocked
#                        - Migratable VMs: live migrate
#                        - Non-migratable VMs: stop, do not block
# None:              VMs powered down (default single-node)

# Configure VM eviction strategy
cat > vm-eviction.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  evictionStrategy: LiveMigrateIfPossible
  template:
    spec:
      domain: {}
EOF

# Edit existing VM
oc edit vm my-vm -n my-namespace
# Update: spec.template.spec.evictionStrategy: LiveMigrateIfPossible

# Restart VM to apply changes
virtctl restart my-vm -n my-namespace

# Cluster-wide eviction strategy
oc edit hyperconverged kubevirt-hyperconverged -n openshift-cnv
# Set: spec.evictionStrategy: LiveMigrate

# Via Web Console:
# 1. Navigate to Virtualization → VirtualMachines
# 2. Select a VM → Policies tab
# 3. Set Eviction Strategy:
#    - LiveMigrate: VMs live migrate during node drain
#    - LiveMigrateIfPossible: Upgrade prioritized, non-migratable VMs stop
#    - None: VMs powered down
# 4. Click Save
# 5. For node affinity/anti-affinity:
#    a. Edit VM → Scheduling
#    b. Configure node selector, affinity, or anti-affinity rules
#    c. For anti-affinity: specify label selector and topologyKey
#    d. Click Save
# 6. For tolerations:
#    a. Edit VM → Scheduling → Tolerations
#    b. Click "Add Toleration"
#    c. Set key, operator, value, effect
#    d. Click Save
# 7. For Machine Health Check (IPI):
#    a. Navigate to Compute → MachineHealthCheck
#    b. Click "Create MachineHealthCheck"
#    c. Configure maxUnhealthy, nodeStartupTimeout, unhealthyConditions
#    d. Click Create

Node Selectors

# Node selector: Simplest form of node placement
cat > vm-node-selector.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  template:
    spec:
      nodeSelector:
        node-role.kubernetes.io/compute: "true"
        disktype: ssd
EOF

# VMs will only schedule on nodes with matching labels

Node Affinity

# Node affinity: More flexible than node selectors

# Required node affinity (hard requirement)
cat > vm-required-affinity.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  template:
    spec:
      affinity:
        nodeAffinity:
          requiredDuringSchedulingIgnoredDuringExecution:
            nodeSelectorTerms:
            - matchExpressions:
              - key: node-role.kubernetes.io/compute
                operator: In
                values:
                - "true"
EOF

# Preferred node affinity (soft requirement)
cat > vm-preferred-affinity.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  template:
    spec:
      affinity:
        nodeAffinity:
          preferredDuringSchedulingIgnoredDuringExecution:
          - weight: 1
            preference:
              matchExpressions:
              - key: node-role.kubernetes.io/compute
                operator: In
                values:
                - "true"
EOF

Anti-Affinity

# Spread VMs across different nodes for high availability
cat > vm-anti-affinity.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  template:
    spec:
      affinity:
        podAntiAffinity:
          requiredDuringSchedulingIgnoredDuringExecution:
          - labelSelector:
              matchExpressions:
              - key: app
                operator: In
                values:
                - my-app
            topologyKey: kubernetes.io/hostname
EOF

# Spread across failure domains
cat > zone-affinity.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  template:
    spec:
      affinity:
        podAntiAffinity:
          requiredDuringSchedulingIgnoredDuringExecution:
          - labelSelector:
              matchExpressions:
              - key: app
                operator: In
                values:
                - my-app
            topologyKey: topology.kubernetes.io/zone
EOF

Tolerations and Taints

# Taints on nodes, tolerations on VMs

# Node with taint
oc adm taint nodes <node_name> key1=value1:NoSchedule

# VM tolerating the taint
cat > vm-toleration.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  template:
    spec:
      tolerations:
      - key: "key1"
        operator: "Equal"
        value: "value1"
        effect: "NoSchedule"
EOF

# Common toleration patterns:
# tolerations:
# - key: "node-role.kubernetes.io/master"
#   operator: "Exists"
#   effect: "NoSchedule"
#
# tolerations:
# - key: "dedicated"
#   operator: "Exists"
#   effect: "NoSchedule"
#
# tolerations:
# - effect: "NoExecute"
#   operator: "Exists"

Watchdog Monitoring

# Watchdog monitors VM health and can trigger node failure recovery

# Configure watchdog for node failure detection
cat > vm-watchdog.yaml <<EOF
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: my-vm
  namespace: my-namespace
spec:
  template:
    spec:
      domain:
        devices:
          watchdogs:
          - name: my-watchdog
            i6300esb:
              action: reset
EOF

# Watchdog actions:
# reset: Reboot VM (default)
# powerOff: Power down VM
# shutdown: Graceful shutdown
# pause: Pause VM

# Health checks for VMs
# - Liveness probe: TCP, HTTP, exec
# - Readiness probe: TCP, HTTP, exec
# - Guest agent ping: Uses QEMU guest agent

High Availability Configuration

# Machine Health Check for IPI clusters
cat > mhc.yaml <<EOF
apiVersion: machine.openshift.io/v1beta1
kind: MachineHealthCheck
metadata:
  name: node-mhc
  namespace: openshift-machine-api
spec:
  clusterSelector: {}
  maxUnhealthy: 20%
  nodeStartupTimeout: 10m
  unhealthyConditions:
  - type: Ready
    status: Unknown
    timeout: 300s
  - type: Ready
    status: "False"
    timeout: 300s
EOF

# Node Health Check Operator for non-IPI
# Deploy Self Node Remediation Operator or Fence Agents Remediation Operator

# Without external monitoring, VMs lose high availability
# Consider: runStrategy: Always with evictionStrategy: LiveMigrate
