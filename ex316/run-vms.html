Run and Access Virtual Machines

Provision and manage VMs from web console or CLI, virtctl commands, user and group access control.

virtctl CLI Tool

# Install virtctl
# On RHEL 9 or later, Linux, Windows, or macOS:
dnf install virtctl

# On RHEL 8:
dnf install virtio-win

# virtctl command categories:

# Information commands
virtctl version
virtctl --help

# VM information commands
virtctl get-vmi <vmi_name> -n <namespace>
virtctl list-vmis -A

# VM manifest creation commands
virtctl create vm --name my-vm --instancetype u1.small --preference rhel.9
virtctl create instancetype --cpu 2 --memory 256Mi

# VM management commands
virtctl start <vm_name> -n <namespace>
virtctl stop <vm_name> -n <namespace>
virtctl restart <vm_name> -n <namespace>
virtctl delete <vm_name> -n <namespace>

# VM connection commands
virtctl console <vm_name> -n <namespace>    # Serial console
virtctl vnc <vm_name> -n <namespace>          # Virtual console
virtctl vga <vm_name> -n <namespace>          # VGA console
virtctl scp <file> <vm_name>:~ -n <namespace>  # Copy file to VM
virtctl cp <host_path> <vm_name>:<vm_path> -n <namespace>  # Copy file to VM
virtctl ssh <user>@<vm_name> -n <namespace>   # SSH to VM
virtctl port-forward <vm_name> 8080:80 -n <namespace>  # Port forward

# VM memory dump commands
virtctl memory-dump <vm_name> -n <namespace>

# Hot plug/unplug commands
virtctl add-volume volume-name <vm_name>
virtctl remove-volume volume-name <vm_name>
virtctl add-device device-name <vm_name>
virtctl remove-device device-name <vm_name>

# Image upload commands
virtctl upload disk --data-volume=my-dv --size=10Gi --image-path=./image.qcow2
virtctl upload iso --data-volume=my-iso-dv --size=2Gi --image-path=./disk.iso

# Deploy libguestfs
virtctl deploy --kubeconfig=~/.kube/config

Instance Types

# Instance types define CPU and memory resources for VMs
apiVersion: instancetype.kubevirt.io/v1beta1
kind: VirtualMachineInstancetype
metadata:
  name: example-instancetype
spec:
  cpu:
    guest: 1
  memory:
    guest: 128Mi
  # Required fields: spec.cpu.guest and spec.memory.guest

# Predefined instance types (common-instancetypes):
#   n1.medium     - 1 vCPU, 2 GiB RAM (general purpose)
#   o1.small      - 1 vCPU, 4 GiB RAM (memory optimized)
#   cx1.2xlarge   - 8 vCPU, 4 GiB RAM (compute optimized)
#   u1.medium     - 2 vCPU, 8 GiB RAM (burstable)
#   m1.large      - 4 vCPU, 4 GiB RAM (standard)

# virtctl usage with instance types
virtctl create vm \
  --instancetype <my_instancetype> \
  --preference <my_preference> \
  --name my-vm

# Specify namespaced instance type
virtctl create vm --instancetype namespace/my-instancetype

# Infer instance type from a running VM
virtctl create vm --name cloned-vm --infer-instancetype <source_vm>

# Creating instance types manually or with virtctl
virtctl create instancetype --cpu 2 --memory 256Mi

Creating VMs from Templates

# CLI: Create VM from template with boot source
virtctl create vm \
  --name vm-rhel-9 \
  --instancetype u1.small \
  --preference rhel.9 \
  --volumeimport type:http,url:https://example.com/rhel9.qcow2,size:10Gi

# VM manifest structure
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: vm-rhel-9
spec:
  dataVolumeTemplates:
  - metadata:
      name: imported-volume
    spec:
      source:
        http:
          url: https://example.com/rhel9.qcow2
      storage:
        resources:
          requests:
            storage: 10Gi
  instancetype:
    name: u1.small
  preference:
    name: rhel.9
  runStrategy: Always
  template:
    spec:
      domain:
        devices: {}
        resources: {}
      terminationGracePeriodSeconds: 180
      volumes:
      - dataVolume:
          name: imported-volume
        name: imported-volume

# Create the VM
oc create -f <vm_manifest>.yaml

# Monitor progress
oc get dv <data_volume_name>  # Should show PHASE: Succeeded
oc get pods                    # View importer pod
virtctl console <vm_name>    # Access serial console

Creating VMs by Uploading Images

# Upload OS images (IMG, ISO, QCOW2) from local machine
# Via Web Console:
# 1. Navigate to Virtualization → Catalog
# 2. Click template without boot source
# 3. Click Customize VirtualMachine
# 4. Under Storage, select "Upload (Upload a new file to a PVC)"
# 5. Browse to image, set disk size
# 6. Customize VM, click Create

# Via CLI - Upload disk to PVC
virtctl upload disk \
  --data-volume=my-dv \
  --size=10Gi \
  --image-path=./my-image.qcow2 \
  -n my-namespace

# Import from URL
cat > dv.yaml <<EOF
apiVersion: cdi.kubevirt.io/v1beta1
kind: DataVolume
metadata:
  name: my-import
  namespace: my-namespace
spec:
  source:
    http:
      url: https://example.com/image.qcow2
  storage:
    resources:
      requests:
        storage: 10Gi
EOF

oc apply -f dv.yaml

# Monitor import progress
oc get dv my-import -n my-namespace

VM Access Control (RBAC)

# Default cluster roles for OpenShift Virtualization
# - kubevirt-snapshot-admin
# - kubevirt-hyperconverged-clusterrole

# RBAC for storage features
# Cluster-wide RBAC roles:
# - hyperconverged-clusters
# - kubevirt-admin
# - kubevirt-edit
# - kubevirt-view

# Namespaced RBAC roles:
# - kubevirt-virtual-machine-admin
# - kubevirt-virtual-machine-user
# - kubevirt-virtual-machine-operator

# Grant a user access to view VMs in a namespace
cat > vm-viewer-rolebinding.yaml <<EOF
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: vm-viewer
  namespace: my-namespace
subjects:
- kind: User
  name: john.doe
  apiGroup: rbac.authorization.k8s.io
roleRef:
  kind: ClusterRole
  name: kubevirt-virtual-machine-user
  apiGroup: rbac.authorization.k8s.io
EOF

oc apply -f vm-viewer-rolebinding.yaml

# Grant admin access to manage VMs in a namespace
cat > vm-admin-rolebinding.yaml <<EOF
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: vm-admin
  namespace: my-namespace
subjects:
- kind: User
  name: jane.admin
  apiGroup: rbac.authorization.k8s.io
roleRef:
  kind: ClusterRole
  name: kubevirt-virtual-machine-admin
  apiGroup: rbac.authorization.k8s.io
EOF

oc apply -f vm-admin-rolebinding.yaml

Accessing Virtual Machines

# Serial console access
virtctl console <vm_name> -n <namespace>
# Escape sequence: ^]

# VNC graphical console
virtctl vnc <vm_name> -n <namespace>

# SSH access
virtctl ssh user@<vm_name> -n <namespace>

# Copy files to/from VM
virtctl cp local_file.txt <vm_name>:~/remote_file.txt -n <namespace>

# Port forwarding
virtctl port-forward <vm_name> 8080:80 -n <namespace>
# Then access: http://localhost:8080

# Install QEMU guest agent on Linux VMs (required for quiesced snapshots)
yum install -y qemu-guest-agent
systemctl enable --now qemu-guest-agent

# Verify agent is connected
oc get vm <vm_name>
# Check for AgentConnected status
