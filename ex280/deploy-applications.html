Deploy Applications

Templates, Helm charts, deployments, replica sets, labels, selectors, services, and external access.

Deploying from Templates

# List available template libraries
oc template --library

# Import a template from a file
oc process -f template.yaml -o yaml

# Process a template with parameters
oc process -f template.yaml -p APP_NAME=myapp -p REPLICAS=3 -o yaml | oc apply -f -

# Process template with parameter file
oc process -f template.yaml -p params.env -o yaml | oc apply -f -

# Process template with environment variables
oc process -f template.yaml -p APP_NAME=myapp -n <project> -o yaml | oc apply -f -

# View template without processing
oc process -f template.yaml --dry-run -o yaml

# Template example
cat > template.yaml <<'EOF'
apiVersion: template.openshift.io/v1
kind: Template
metadata:
  name: my-app-template
objects:
- apiVersion: apps/v1
  kind: Deployment
  metadata:
    name: ${APP_NAME}
  spec:
    replicas: ${REPLICAS:-1}
    selector:
      matchLabels:
        app: ${APP_NAME}
    template:
      metadata:
        labels:
          app: ${APP_NAME}
      spec:
        containers:
        - name: ${APP_NAME}
          image: ${IMAGE:-nginx:latest}
          ports:
          - containerPort: ${PORT:-80}
parameters:
- name: APP_NAME
  value: my-app
  required: true
- name: REPLICAS
  value: "1"
  required: false
- name: IMAGE
  value: nginx:latest
  required: false
- name: PORT
  value: "80"
  required: false
EOF

Deploying from Helm Charts

# Install Helm (if not available)
sudo dnf install -y helm

# Add a Helm repository
helm repo add bitnami https://charts.bitnami.com/bitnami
helm repo update

# Search for charts
helm search repo nginx

# Install a Helm chart
helm install my-app bitnami/nginx --namespace my-namespace --create-namespace

# Install with custom values
helm install my-app bitnami/nginx -f values.yaml --namespace my-namespace

# Install from a local chart
helm install my-app ./my-chart --namespace my-namespace

# List Helm releases
helm list -n my-namespace
helm list --all-namespaces

# View Helm release status
helm status my-app -n my-namespace

# View Helm release history
helm history my-app -n my-namespace

# Upgrade a Helm release
helm upgrade my-app bitnami/nginx -f values.yaml

# Rollback a Helm release
helm rollback my-app 1 -n my-namespace

# Uninstall a Helm release
helm uninstall my-app -n my-namespace

# Generate Kubernetes manifests from Helm (without installing)
helm template my-app bitnami/nginx -f values.yaml > manifests.yaml

# Lint a Helm chart
helm lint ./my-chart

# Package a Helm chart
helm package ./my-chart

# Create Chart.yaml
cat > Chart.yaml <<'EOF'
apiVersion: v2
name: my-chart
description: A custom Helm chart
version: 1.0.0
appVersion: "1.0"
EOF

# Create values.yaml
cat > values.yaml <<'EOF'
replicaCount: 1

image:
  repository: nginx
  tag: "1.21"
  pullPolicy: IfNotPresent

service:
  type: ClusterIP
  port: 80
EOF

Managing Application Deployments

# Scale a deployment
oc scale deployment/<deploy-name> --replicas=5
oc scale deployment/<deploy-name> --replicas=0  # Scale down

# Scale using resources.yaml
oc apply -f deployment.yaml

# Rollout status
oc rollout status deployment/<deploy-name>

# Rollback
oc rollout undo deployment/<deploy-name>
oc rollout undo deployment/<deploy-name> --to-revision=2

# History
oc rollout history deployment/<deploy-name>

# Restart
oc rollout restart deployment/<deploy-name>

# Pause/Resume rollout
oc rollout pause deployment/<deploy-name>
oc rollout resume deployment/<deploy-name>

Working with Replica Sets

# ReplicaSets are managed by Deployments - avoid managing directly
oc get replicasets -n <project>

# View replica set details
oc describe rs <rs-name> -n <project>

# Create a ReplicaSet (not recommended for normal use)
cat > replicaset.yaml <<'EOF'
apiVersion: apps/v1
kind: ReplicaSet
metadata:
  name: my-app-rs
spec:
  replicas: 3
  selector:
    matchLabels:
      app: my-app
  template:
    metadata:
      labels:
        app: my-app
    spec:
      containers:
      - name: my-app
        image: nginx:1.21
EOF

oc apply -f replicaset.yaml

# Scale a replica set
oc scale rs/<rs-name> --replicas=5

# Note: Directly managing ReplicaSets bypasses Deployment strategy
# Use Deployments for controlled scaling and rollouts

Working with Labels and Selectors

# Label a resource
oc label deployment/<deploy-name> env=prod
oc label deployment/<deploy-name> tier=frontend

# Update an existing label
oc label deployment/<deploy-name> env=staging --overwrite

# Remove a label
oc label deployment/<deploy-name> env-

# Apply labels to multiple resources
oc label deployment -l app=myapp env=prod --overwrite

# Label based on a file
oc label -f resources.yaml env=prod --overwrite

# Select by label
oc get pods -l app=myapp
oc get pods -l 'app in (web, api)'
oc get pods -l 'app notin (test)'

# Field selectors
oc get pods --field-selector=status.phase=Running
oc get pods --field-selector=status.phase=Failed

# Common label conventions
#   app.kubernetes.io/name
#   app.kubernetes.io/instance
#   app.kubernetes.io/version
#   app.kubernetes.io/component
#   app.kubernetes.io/part-of
#   app.kubernetes.io/managed-by
#   app.kubernetes.io/environment

oc get pods -l 'app.kubernetes.io/environment=production'

Configuring Services

# ClusterIP (default - internal only)
cat > svc-clusterip.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: my-clusterip-svc
spec:
  type: ClusterIP
  selector:
    app: my-app
  ports:
  - port: 80
    targetPort: 8080
    protocol: TCP
EOF

# NodePort (exposes on each node IP)
cat > svc-nodeport.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: my-nodeport-svc
spec:
  type: NodePort
  selector:
    app: my-app
  ports:
  - port: 80
    targetPort: 8080
    nodePort: 30080
EOF

# LoadBalancer (cloud provider)
cat > svc-lb.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: my-lb-svc
spec:
  type: LoadBalancer
  selector:
    app: my-app
  ports:
  - port: 80
    targetPort: 8080
EOF

# Headless service (no load balancing)
cat > svc-headless.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: my-headless-svc
spec:
  clusterIP: None
  selector:
    app: my-app
  ports:
  - port: 80
    targetPort: 8080
EOF

# Apply services
oc apply -f svc-clusterip.yaml

# Port configuration options
cat > svc-ports.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: my-multiport-svc
spec:
  selector:
    app: my-app
  ports:
  - name: http
    port: 80
    targetPort: 8080
    nodePort: 30080
  - name: https
    port: 443
    targetPort: 8443
    protocol: TCP
  - name: metrics
    port: 9090
    targetPort: 9090
    protocol: TCP
  type: ClusterIP
EOF

# View service endpoints
oc get endpoints <svc-name> -n <project>

# Port naming convention: [protocol:]name[-port]
#   TCP/UDP is default, must specify HTTP/HTTPS explicitly
oc get svc --all-namespaces

Exposing Applications to External Access

# Expose a deployment as a Route (OpenShift)
oc expose deployment/<deploy-name> --port=80 --name=my-route

# Create a Route manually
cat > route.yaml <<'EOF'
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: my-app-route
spec:
  to:
    kind: Service
    name: my-app-service
    weight: 100
  port:
    targetPort: 8080
  tls:
    termination: edge
    insecureEdgeTerminationPolicy: Redirect
EOF

oc apply -f route.yaml

# Route types
#   edge: TLS terminates at router (default)
#   reencrypt: Router decrypts and re-encrypts to backend
#   passthrough: Router passes traffic through (backend handles TLS)

# Create a route with TLS
cat > route-tls.yaml <<'EOF'
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: my-tls-route
spec:
  host: myapp.example.com
  to:
    kind: Service
    name: my-app-service
    weight: 100
  port:
    targetPort: 8443
  tls:
    termination: reencrypt
    key: |
      -----BEGIN RSA PRIVATE KEY-----
      ...
    certificate: |
      -----BEGIN CERTIFICATE-----
      ...
    caCertificate: |
      -----BEGIN CERTIFICATE-----
      ...
    destinationCACertificate: |
      -----BEGIN CERTIFICATE-----
      ...
EOF

# Create a route with passthrough
cat > route-passthrough.yaml <<'EOF'
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: my-passthrough-route
spec:
  host: myapp.example.com
  to:
    kind: Service
    name: my-app-service
    weight: 100
  tls:
    termination: passthrough
EOF

# View routes
oc get routes -n <project>
oc get routes -o wide

# Expose a service as a Route
oc create route edge --service=my-app-svc --hostname=myapp.example.com
