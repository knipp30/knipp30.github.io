Manage Authentication & Authorization

HTPasswd identity provider, user and group management, password modification, and permissions.

HTPasswd Identity Provider

# Create an htpasswd file using htpasswd utility
htpasswd -bBC 10 -c /tmp/htpasswd admin mypassword

# Add more users
htpasswd -bB /tmp/htpasswd user1 password1
htpasswd -bB /tmp/htpasswd user2 password2

# Create the HTPasswd Secret in openshift-config namespace
oc create secret generic htpass-secret --from-file=htpasswd=/tmp/htpasswd -n openshift-config

# Create the HTPasswd provider configmap
cat > htpasswd-config.yaml <<'EOF'
apiVersion: config.openshift.io/v1
kind: OAuth
metadata:
  name: cluster
spec:
  identityProviders:
  - name: htpasswd-provider
    mappingMethod: claim
    type: HTPasswd
    htpasswd:
      fileData:
        name: htpass-secret
EOF

oc apply -f htpasswd-config.yaml

# Verify identity provider configuration
oc get oauth cluster -o yaml

# Test login with HTPasswd
oc login -u admin -p mypassword https://<cluster-api>:6443

Creating and Managing Users

# Create a user (resource)
cat > user.yaml <<'EOF'
apiVersion: user.openshift.io/v1
kind: User
metadata:
  name: myuser@example.com
  labels:
    example: "true"
# Note: Users are identity records, not Kubernetes resources
EOF

# Users are typically created via identity providers (HTPasswd, LDAP, etc.)
# Verify users
oc get identity

# Verify user identity mapping
oc get oauth

# View user details
oc whoami
oc whoami --show-groups
oc whoami --show-cert

# Login as a user
oc login -u myuser@example.com -p mypassword https://<cluster-api>:6443

# Check what resources a user can access
oc auth can-i create pods -n my-project
oc auth can-i create deployments -n my-project
oc auth can-i list pods --as=myuser@example.com

Modifying User Passwords

# For HTPasswd users: update the htpasswd file and re-create the secret
htpasswd -bB /tmp/htpasswd user1 newpassword

# Delete the old secret and create a new one
oc delete secret htpass-secret -n openshift-config
oc create secret generic htpass-secret --from-file=htpasswd=/tmp/htpasswd -n openshift-config

# Password must persist after reboot - keep the htpasswd file in a persistent location
# Store it on the master node or in a persistent volume
# In OCP 4.x, the secret is stored in the openshift-config namespace

# Verify
oc login -u user1 -p newpassword https://<cluster-api>:6443

Creating and Managing Groups

# Create a group
cat > group.yaml <<'EOF'
apiVersion: user.openshift.io/v1
kind: Group
metadata:
  name: developers
users:
- alice@example.com
- bob@example.com
EOF

oc apply -f group.yaml

# List groups
oc get groups

# View group members
oc describe group developers

# Add a user to a group
oc adm groups add-users developers alice@example.com bob@example.com

# Remove a user from a group
oc adm groups remove-users developers alice@example.com

# List groups for a user
oc groups alice@example.com

# List all users in a group
oc get group developers -o yaml

# Delete a group
oc delete group developers

User and Group Permissions

# Roles and RoleBindings
cat > role.yaml <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  namespace: my-project
  name: pod-viewer
rules:
- apiGroups: [""]
  resources: ["pods"]
  verbs: ["get", "list", "watch"]
EOF

oc apply -f role.yaml

# Bind a role to a user
cat > rolebinding.yaml <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: pod-viewer-binding
  namespace: my-project
subjects:
- kind: User
  name: alice@example.com
  apiGroup: rbac.authorization.k8s.io
roleRef:
  kind: Role
  name: pod-viewer
  apiGroup: rbac.authorization.k8s.io
EOF

oc apply -f rolebinding.yaml

# Bind a role to a group
cat > groupbinding.yaml <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: developers-binding
  namespace: my-project
subjects:
- kind: Group
  name: developers
  apiGroup: rbac.authorization.k8s.io
roleRef:
  kind: Role
  name: pod-viewer
  apiGroup: rbac.authorization.k8s.io
EOF

oc apply -f groupbinding.yaml

# Check permissions
oc auth can-i create pods -n my-project --as=alice@example.com
oc auth can-i create deployments -n my-project --as=alice@example.com

# Cluster roles and bindings (cluster-wide)
cat > clusterrole.yaml <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRole
metadata:
  name: node-reader
rules:
- apiGroups: [""]
  resources: ["nodes"]
  verbs: ["get", "list", "watch"]
EOF

oc apply -f clusterrole.yaml

cat > clusterrolebinding.yaml <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: node-readers
subjects:
- kind: Group
  name: developers
  apiGroup: rbac.authorization.k8s.io
roleRef:
  kind: ClusterRole
  name: node-reader
  apiGroup: rbac.authorization.k8s.io
EOF

oc apply -f clusterrolebinding.yaml

# Predefined roles
#   admin: Full access to namespace
#   edit: Create/update/delete resources (cannot modify RBAC)
#   view: Read-only access to resources
oc adm policy add-cluster-role-to-user cluster-admin admin@example.com
oc adm policy add-cluster-role-to-group cluster-admin developers

Self-Provisioner ClusterRole

# The self-provisioner ClusterRole allows users to create their own projects
# Enable/disable self-provisioning
oc adm policy remove-cluster-role-from-group self-provisioner system:authenticated:oauth
oc adm policy add-cluster-role-to-group self-provisioner system:authenticated:oauth

# View self-provisioner bindings
oc get clusterrolebinding self-provisioners
