Configure Application Security

Service accounts, privileged apps, SCCs, secrets, Kubernetes API access, jobs, and cron jobs.

Overview

Application security in OpenShift involves managing identities, permissions, access controls, and workload isolation to ensure secure application deployment.

Configuring and Managing Service Accounts

# View the default service account
oc describe serviceaccount default -n my-project

# Create a custom service account
cat > sa.yaml <<'EOF'
apiVersion: v1
kind: ServiceAccount
metadata:
  name: my-app-sa
  namespace: my-project
EOF

oc apply -f sa.yaml

# Use a service account in a pod
cat > pod-sa.yaml <<'EOF'
apiVersion: v1
kind: Pod
metadata:
  name: my-pod
spec:
  serviceAccountName: my-app-sa
  containers:
  - name: my-app
    image: nginx
EOF

oc apply -f pod-sa.yaml

# Create a service account with image pull secrets
cat > sa-pull-secret.yaml <<'EOF'
apiVersion: v1
kind: ServiceAccount
metadata:
  name: my-app-sa
  namespace: my-project
imagePullSecrets:
- name: registry-creds
EOF

oc apply -f sa-pull-secret.yaml

# Link a service account to an image pull secret
oc secrets link my-app-sa registry-creds --for=pull -n my-project

# Automount service account token (security best practice)
cat > sa-no-automount.yaml <<'EOF'
apiVersion: v1
kind: ServiceAccount
metadata:
  name: my-app-sa
  namespace: my-project
automountServiceAccountToken: false
EOF

oc apply -f sa-no-automount.yaml

# Use a service account with explicit token mount
cat > pod-token-mount.yaml <<'EOF'
apiVersion: v1
kind: Pod
metadata:
  name: my-pod
spec:
  serviceAccountName: my-app-sa
  automountServiceAccountToken: false
  containers:
  - name: my-app
    image: nginx
    volumeMounts:
    - name: token
      mountPath: /var/run/secrets/tokens
      readOnly: true
  volumes:
  - name: token
    projected:
      sources:
      - serviceAccountToken:
          path: token
          audience: openshift
          expirationSeconds: 3600
EOF

oc apply -f pod-token-mount.yaml

Running Privileged Applications

# Run a privileged container (requires elevated privileges)
cat > privileged-pod.yaml <<'EOF'
apiVersion: v1
kind: Pod
metadata:
  name: privileged-pod
spec:
  containers:
  - name: privileged-container
    image: nginx
    securityContext:
      privileged: true
      runAsUser: 0
EOF

# Apply requires SecurityContextConstraints approval
oc apply -f privileged-pod.yaml

# View SCC requirements
oc get scc

# Common SCCs:
#   privileged: Full host access
#   hostnetwork: Access to host network
#   hostaccess: Access to host filesystem
#   restricted: Default - no root, no privileged

# Use a specific SCC (must be in the service account's allowed SCCs)
cat > scc-pod.yaml <<'EOF'
apiVersion: v1
kind: Pod
metadata:
  name: my-pod
spec:
  securityContext:
    runAsUser: 0
    seLinuxOptions:
      level: s0:c123,c456
  containers:
  - name: my-app
    image: nginx
    securityContext:
      privileged: true
      runAsUser: 0
EOF

oc apply -f scc-pod.yaml

Managing Security Context Constraints

# View available SCCs
oc get scc
oc describe scc restricted
oc describe scc privileged

# SCC fields:
#   allowPrivilegedContainer: Allow privileged containers
#   runAsUser: User ID strategy (MustRunAs, RunAsAny, etc.)
#   fsGroup: Filesystem group strategy
#   seLinuxContext: SELinux strategy
#   supplementalGroups: Supplemental group strategy
#   volumes: Allowed volume types
#   allowedCapabilities: Allowed capabilities

# Add a user to the privileged SCC (requires cluster-admin)
oc adm policy add-scc-to-user privileged -z my-app-sa -n my-project
oc adm policy add-scc-to-user privileged system:serviceaccount:my-project:my-app-sa

# Add SCC to a group
oc adm policy add-scc-to-group privileged system:authenticated:oauth

# Remove SCC from a user
oc adm policy remove-scc-from-user privileged -z my-app-sa -n my-project

# SCC binding
cat > scc-binding.yaml <<'EOF'
apiVersion: authorization.openshift.io/v1
kind: SubjectAccessReview
metadata:
  name: scc-binding
spec:
  resources:
  - verbs: ["use"]
    resource: securitycontextconstraints
    name: privileged
  user: system:serviceaccount:my-project:my-app-sa
EOF

Managing and Applying Secrets

# Create a secret
oc create secret generic db-credentials \
  --from-literal=username=admin \
  --from-literal=password=secret123 \
  -n my-project

# Create a TLS secret
oc create secret tls my-tls --cert=tls.crt --key=tls.key -n my-project

# Create a dockercfg secret
oc create secret docker-registry registry-pull-secret \
  --docker-server=registry.example.com \
  --docker-username=pulluser \
  --docker-password=pullpass \
  --docker-email=admin@example.com \
  -n my-project

# Use secrets in a pod
cat > pod-secret.yaml <<'EOF'
apiVersion: v1
kind: Pod
metadata:
  name: my-pod
spec:
  containers:
  - name: my-app
    image: nginx
    env:
    - name: DB_USER
      valueFrom:
        secretKeyRef:
          name: db-credentials
          key: username
    - name: DB_PASS
      valueFrom:
        secretKeyRef:
          name: db-credentials
          key: password
    volumeMounts:
    - name: tls-secret
      mountPath: /etc/tls
      readOnly: true
  volumes:
  - name: tls-secret
    secret:
      secretName: my-tls
EOF

oc apply -f pod-secret.yaml

# Rotate secrets by updating the secret and restarting pods
oc set env deployment/my-app DB_PASS=newpassword --from=secret/db-credentials
oc rollout restart deployment/my-app

Configuring Application Access to Kubernetes APIs

# Service accounts get automatic access to kube-apiserver
# Token is mounted at /var/run/secrets/kubernetes.io/serviceaccount/token

# Use the service account token to access Kubernetes API
cat > api-access.yaml <<'EOF'
apiVersion: v1
kind: Pod
metadata:
  name: api-test
spec:
  serviceAccountName: my-app-sa
  containers:
  - name: api-test
    image: curlimages/curl
    command: ['sh', '-c', 'while true; do curl -s -k --cacert /var/run/secrets/kubernetes.io/serviceaccount/ca.crt --header "Authorization: Bearer $(cat /var/run/secrets/kubernetes.io/serviceaccount/token)" https://kubernetes.default.svc:443/api/v1/namespaces/my-project/pods; sleep 30; done']
    volumeMounts:
    - name: ca-cert
      mountPath: /var/run/secrets/kubernetes.io/serviceaccount
      readOnly: true
  volumes:
  - name: ca-cert
    projected:
      sources:
      - serviceAccountToken:
          audience: api
          expirationSeconds: 3600
          path: token
      - configMap:
          name: kube-root-ca.crt
          items:
          - key: ca.crt
            path: ca.crt
EOF

oc apply -f api-access.yaml

# Use RBAC to grant API access
cat > api-role.yaml <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  name: pod-reader
  namespace: my-project
rules:
- apiGroups: [""]
  resources: ["pods"]
  verbs: ["get", "list", "watch"]
- apiGroups: ["apps"]
  resources: ["deployments"]
  verbs: ["get", "list", "watch"]
EOF

oc apply -f api-role.yaml

cat > api-rolebinding.yaml <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: pod-reader-binding
  namespace: my-project
subjects:
- kind: ServiceAccount
  name: my-app-sa
  namespace: my-project
roleRef:
  kind: Role
  name: pod-reader
  apiGroup: rbac.authorization.k8s.io
EOF

oc apply -f api-rolebinding.yaml

Deploying Jobs for One-Time Tasks

# Create a simple Job
cat > job.yaml <<'EOF'
apiVersion: batch/v1
kind: Job
metadata:
  name: my-job
spec:
  template:
    spec:
      containers:
      - name: my-job-container
        image: busybox
        command: ['sh', '-c', 'echo "Running one-time task" && sleep 5']
      restartPolicy: Never
  backoffLimit: 4
EOF

oc apply -f job.yaml

# Create a Job with retries and parallelism
cat > job-parallel.yaml <<'EOF'
apiVersion: batch/v1
kind: Job
metadata:
  name: parallel-job
spec:
  parallelism: 3
  completions: 6
  template:
    spec:
      containers:
      - name: parallel-task
        image: busybox
        command: ['sh', '-c', 'echo "Processing task $JOB_COMPLETION_INDEX" && sleep 10']
      restartPolicy: Never
  backoffLimit: 2
  completionMode: Indexed
EOF

oc apply -f job-parallel.yaml

# View job status
oc get jobs -n my-project
oc get jobs -n my-project -o wide
oc describe job/my-job -n my-project

# View job logs
oc logs job/my-job
oc logs job/my-job -f

# View pod logs for a specific job
oc get pods -l job-name=my-job -n my-project
oc logs <pod-name-from-job> -n my-project

# Delete a job
oc delete job/my-job -n my-project

# Create a job from imperative commands
oc run my-job --image=busybox --restart=Never --command -- echo "Hello from job"

oc create job my-job --from=cronjob/my-cronjob

Configuring Kubernetes Cron Jobs

# Create a CronJob
cat > cronjob.yaml <<'EOF'
apiVersion: batch/v1
kind: CronJob
metadata:
  name: my-cronjob
spec:
  schedule: "*/5 * * * *"  # Every 5 minutes
  # Cron schedule format: minute hour day-of-month month day-of-week
  # Examples:
  #   "0 * * * *"       - Every hour
  #   "0 0 * * *"       - Daily at midnight
  #   "0 0 * * 0"       - Weekly on Sunday at midnight
  #   "0 */2 * * *"     - Every 2 hours
  #   "30 2 * * 1"      - Monday at 2:30 AM
  successfulJobsHistoryLimit: 3
  failedJobsHistoryLimit: 1
  concurrencyPolicy: Forbid  # or "Allow" or "Replace"
  jobTemplate:
    spec:
      template:
        spec:
          containers:
          - name: cron-container
            image: busybox
            command: ['sh', '-c', 'echo "Running scheduled task" && date']
          restartPolicy: OnFailure
          backoffLimit: 2
EOF

oc apply -f cronjob.yaml

# View CronJob status
oc get cronjob -n my-project
oc describe cronjob/my-cronjob -n my-project

# View associated jobs
oc get jobs -n my-project -l cronjob=my-cronjob

# View job execution history
oc get jobs -n my-project --sort-by='.metadata.creationTimestamp'

# Manually create a Job from a CronJob
oc create job manual-job --from=cronjob/my-cronjob

# Pause and resume a CronJob
oc patch cronjob/my-cronjob -p '{"spec":{"suspend":true}}' -n my-project
oc patch cronjob/my-cronjob -p '{"spec":{"suspend":false}}' -n my-project

# Delete a CronJob
oc delete cronjob/my-cronjob -n my-project

Security Best Practices

# Always use non-root users in containers
# Set security context at pod or container level
cat > secure-pod.yaml <<'EOF'
apiVersion: v1
kind: Pod
metadata:
  name: secure-pod
spec:
  securityContext:
    runAsNonRoot: true
    runAsUser: 1000
    runAsGroup: 3000
    fsGroup: 2000
    seccompProfile:
      type: RuntimeDefault
  containers:
  - name: app
    image: nginx
    securityContext:
      allowPrivilegeEscalation: false
      capabilities:
        drop:
        - ALL
      readOnlyRootFilesystem: true
    volumeMounts:
    - name: tmp
      mountPath: /tmp
    - name: cache
      mountPath: /var/cache/nginx
      - name: run
        mountPath: /var/run/nginx
  volumes:
  - name: tmp
    emptyDir: {}
  - name: cache
    emptyDir: {}
  - name: run
    emptyDir: {}
EOF

oc apply -f secure-pod.yaml

# Check what SCC a service account can use
oc who-can use scc privileged -n my-project
