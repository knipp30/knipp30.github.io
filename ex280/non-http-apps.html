Expose Non-HTTP/SNI Applications

Load balancer services, external application access, TCP/UDP workload routing.

Overview

OpenShift Routes are optimized for HTTP/HTTPS and SNI-based traffic. For TCP/UDP applications (databases, Redis, game servers, etc.), you need alternative approaches.

Configuring a Load Balancer Service

# Create a LoadBalancer Service
cat > loadbalancer-svc.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: my-loadbalancer-svc
  annotations:
    service.beta.kubernetes.io/aws-load-balancer-type: "nlb"
spec:
  type: LoadBalancer
  selector:
    app: my-app
  ports:
  - name: tcp
    port: 8443
    targetPort: 8443
    protocol: TCP
EOF

oc apply -f loadbalancer-svc.yaml

# View LoadBalancer service
oc get svc/my-loadbalancer-svc -o wide

# LoadBalancer external IP (may take time to provision)
oc get svc/my-loadbalancer-svc

# Create a LoadBalancer with specific ports
cat > multiport-lb.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: multiport-lb
spec:
  type: LoadBalancer
  selector:
    app: my-app
  ports:
  - name: tcp-primary
    port: 8080
    targetPort: 8080
    protocol: TCP
  - name: tcp-secondary
    port: 8443
    targetPort: 8443
    protocol: TCP
  - name: udp-monitoring
    port: 9090
    targetPort: 9090
    protocol: UDP
EOF

oc apply -f multiport-lb.yaml

Configuring External Application Access

# Method 1: NodePort Service (access via node IP + port)
cat > nodeport-svc.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: my-nodeport-svc
spec:
  type: NodePort
  selector:
    app: my-app
  ports:
  - name: tcp
    port: 8080
    targetPort: 8080
    nodePort: 30080  # Optional: specify a port in range 30000-32767
EOF

oc apply -f nodeport-svc.yaml

# Access from outside: http://<node-ip>:30080
oc get svc/my-nodeport-svc
oc get endpoints/my-nodeport-svc

# Method 2: Route with TLS passthrough (for TCP/HTTPS)
cat > route-passthrough.yaml <<'EOF'
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: my-tcp-route
spec:
  host: my-tcp.example.com
  to:
    kind: Service
    name: my-nodeport-svc
    weight: 100
  port:
    targetPort: 8443
  tls:
    termination: passthrough
EOF

oc apply -f route-passthrough.yaml

# Method 3: HostPort (direct binding to node - use with caution)
cat > pod-hostport.yaml <<'EOF'
apiVersion: v1
kind: Pod
metadata:
  name: my-pod
spec:
  hostNetwork: true
  containers:
  - name: my-app
    image: nginx
    ports:
    - containerPort: 8080
      hostPort: 8080
EOF

# Method 4: ExternalName service (DNS-based external access)
cat > externalname-svc.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: my-external-name-svc
spec:
  type: ExternalName
  externalName: api.external-service.com
EOF

oc apply -f externalname-svc.yaml

TCP/UDP Service Configuration

# TCP Service
cat > tcp-svc.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: tcp-service
spec:
  type: ClusterIP
  selector:
    app: my-tcp-app
  ports:
  - name: tcp-app
    port: 5432
    targetPort: 5432
    protocol: TCP
EOF

# UDP Service
cat > udp-svc.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: udp-service
spec:
  type: ClusterIP
  selector:
    app: my-udp-app
  ports:
  - name: udp-app
    port: 5353
    targetPort: 5353
    protocol: UDP
EOF

# Mixed TCP/UDP Service
cat > mixed-svc.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: mixed-service
spec:
  type: ClusterIP
  selector:
    app: my-app
  ports:
  - name: tcp
    port: 8080
    targetPort: 8080
    protocol: TCP
  - name: udp
    port: 8080
    targetPort: 8080
    protocol: UDP
EOF

# Verify services
oc get svc tcp-service udp-service mixed-service -n my-project
oc get endpoints tcp-service -n my-project

Non-HTTP/SNI with TLS

# For TLS-terminated TCP services, use NodePort or LoadBalancer
# with the application handling TLS termination

# Example: TLS-terminated TCP service
cat > tls-tcp-svc.yaml <<'EOF'
apiVersion: v1
kind: Service
metadata:
  name: tls-tcp-svc
spec:
  type: NodePort
  selector:
    app: tls-tcp-app
  ports:
  - name: tls
    port: 8443
    targetPort: 8443
    protocol: TCP
    nodePort: 30443
EOF

oc apply -f tls-tcp-svc.yaml

# Verify the service and endpoints
oc get svc tls-tcp-svc
oc get endpoints tls-tcp-svc

# Test connectivity
nc -zv <node-ip> 30443
