Configure Network Security

Networking components, SDN troubleshooting, external routes, ingress control, TLS, and network policies.

Configure Networking Components

# View network operator status
oc get co networking

# Check network configuration
oc get network.operator cluster -o yaml

# Check default network configuration
oc get network.operator cluster -o jsonpath='{.status}'

# Network types
#   OVN-Kubernetes: Recommended and default SDN for OCP 4.13+

# Check SDN version
oc get clusterversion
oc get network.operator cluster -o yaml | grep networkType

# Configure SDN MTU
cat > network-config.yaml <<'EOF'
apiVersion: network.operator.openshift.io/v1
kind: Network
metadata:
  name: cluster
spec:
  defaultNetwork:
    ovnKubernetes:
      mtu: 1450
  networkType: OVNKubernetes
EOF

oc apply -f network-config.yaml

Troubleshooting Software Defined Networking

⚠️ Important

OpenShift 4.x uses OpenShiftSDN (default in OCP 4.0-4.12) or OVN-Kubernetes (default since OCP 4.13). SCTP protocol is NOT supported by either default SDN.

# Check SDN pods
oc get pods -n openshift-sdn
oc get pods -n openshift-ovn-kubernetes

# Check SDN services
oc get svc -n openshift-sdn

# Check network operator logs
oc logs -n openshift-sdn daemonset/openshift-sdn

# Check node-level networking
oc debug node/<node-name>
# chroot /host
# ip addr show
# ip route show
# ip neighbor show
# iptables -L -n -v
# ipvsadm -L -n
# ping <pod-ip>  # from inside the debug pod

# Check OVN pods and logs
oc get pods -n openshift-ovn-kubernetes
oc logs -n openshift-ovn-kubernetes daemonset/ovn-kubernetes

# OVN troubleshooting
oc debug node/<node-name>
# chroot /host
# ovn-nbctl show
# ovn-nbctl ls-list
# ovn-sbctl list ports
# ovn-sbctl find logical-router

Creating and Editing External Routes

# Create a route
cat > route.yaml <<'EOF'
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: my-app
  namespace: my-project
spec:
  host: my-app.example.com
  to:
    kind: Service
    name: my-app-service
    weight: 100
  port:
    targetPort: 8080
EOF

oc apply -f route.yaml

# Edit an existing route
oc edit route/my-app -n my-project

# Change route host
oc patch route/my-app -n my-project -p '{"spec":{"host":"new-host.example.com"}}'

# View route details
oc describe route/my-app -n my-project

# Route inspection
oc get route -n my-project -o wide

# Wildcard routes (cluster-wide default)
# A wildcard subdomain is set at cluster install time
# Access: https://<route-host>.<wildcard-domain>

Controlling Cluster Network Ingress

# Ingress controller configuration
oc get ingresscontroller -n openshift-ingress-operator
oc get ingresscontrollers.config.openshift.io/cluster -o yaml

# Custom ingress controller
cat > ingresscontroller.yaml <<'EOF'
apiVersion: operator.openshift.io/v1
kind: IngressController
metadata:
  name: custom-ingress
  namespace: openshift-ingress-operator
spec:
  domain: custom.example.com
  namespaceSelector:
    matchLabels:
      ingress-controller: "true"
  endpointPublishingStrategy:
    type: HostNetwork
    hostNetwork:
      port: 8080
      protocol: TCP
EOF

# OpenShift Ingress Operator (default)
oc get pods -n openshift-ingress-operator

# Manage default ingress controller
oc edit ingresscontrollers.config.operator.openshift.io/cluster -n openshift-ingress-operator

Securing Traffic with TLS Certificates

# Create a TLS certificate
openssl req -x509 -newkey rsa:2048 -keyout tls.key -out tls.crt -days 365 -nodes -subj "/CN=myapp.example.com"

# Create a TLS secret
oc create secret tls my-tls-secret --cert=tls.crt --key=tls.key -n my-project

# Use TLS in a Route
cat > route-tls.yaml <<'EOF'
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: my-tls-route
spec:
  host: myapp.example.com
  to:
    kind: Service
    name: my-app-service
  tls:
    termination: reencrypt
    key: |
      -----BEGIN RSA PRIVATE KEY-----
      ...
    certificate: |
      -----BEGIN CERTIFICATE-----
      ...
    destinationCACertificate: |
      -----BEGIN CERTIFICATE-----
      ...
EOF

oc apply -f route-tls.yaml

# TLS Termination Types:
#   edge: Router terminates TLS, HTTP to backend (default)
#   reencrypt: Router terminates TLS, re-encrypts to backend
#   passthrough: Router passes encrypted traffic through (backend terminates)
#   none: No TLS

# Configure TLS on Ingress Controller (cluster-wide)
# The default IngressController is named 'cluster' in 'openshift-ingress-operator'
# Edit the default IngressController to apply TLS settings:
oc edit ingresscontrollers.config.operator.openshift.io/cluster -n openshift-ingress-operator

# Or patch with CLI:
oc patch ingresscontrollers.config.operator.openshift.io/cluster -n openshift-ingress-operator -p '{"spec":{"tlsConfiguration":{"minTLSVersion":"VersionTLS12","cipherSuites":["TLS_ECDHE_RSA_WITH_AES_128_GCM_SHA256","TLS_ECDHE_RSA_WITH_AES_256_GCM_SHA384]}}}' --type=merge

# View cluster-wide ingress settings
oc get ingress.config/cluster -o yaml -n openshift-ingress-operator

# View ingress controller pods
oc get pods -n openshift-ingress-operator
oc get pods -n openshift-ingress

Configuring Application Network Policies

# Default: All pods can communicate with all pods (unless policies are applied)

# Create a NetworkPolicy to restrict access
cat > networkpolicy.yaml <<'EOF'
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: deny-all-ingress
  namespace: my-project
spec:
  podSelector: {}
  policyTypes:
  - Ingress
EOF

# Create a NetworkPolicy allowing specific access
cat > networkpolicy-allow.yaml <<'EOF'
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: allow-web
  namespace: my-project
spec:
  podSelector:
    matchLabels:
      app: web
  policyTypes:
  - Ingress
  - Egress
  ingress:
  - from:
    - podSelector:
        matchLabels:
          app: frontend
    ports:
    - protocol: TCP
      port: 8080
  egress:
  - to:
    - podSelector:
        matchLabels:
          app: database
    ports:
    - protocol: TCP
      port: 5432
EOF

oc apply -f networkpolicy-allow.yaml

# Multiple policy types
cat > networkpolicy-multi.yaml <<'EOF'
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: web-api-policy
  namespace: my-project
spec:
  podSelector:
    matchLabels:
      app: web-api
  policyTypes:
  - Ingress
  ingress:
  - from:
    - namespaceSelector:
        matchLabels:
          name: frontend
      podSelector:
        matchLabels:
          role: web
    - ipBlock:
        cidr: 10.0.0.0/8
    ports:
    - protocol: TCP
      port: 8080
    - protocol: TCP
      port: 8443
EOF

oc apply -f networkpolicy-multi.yaml

# View network policies
oc get networkpolicy -n my-project
oc describe networkpolicy/web-api-policy -n my-project
