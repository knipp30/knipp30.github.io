Enable Developer Self-Service

Resource quotas, limits, requirements, and project templates for developer productivity.

Overview

Resource quotas and limits allow cluster administrators to control resource consumption per project while enabling developers to self-service their workloads.

Configuring Cluster Resource Quotas

# ResourceQuota in a project
cat > resourcequota.yaml <<'EOF'
apiVersion: v1
kind: ResourceQuota
metadata:
  name: project-quota
spec:
  hard:
    requests.cpu: "4"
    requests.memory: 8Gi
    limits.cpu: "8"
    limits.memory: 16Gi
    pods: "20"
    services: "10"
    persistentvolumeclaims: "10"
    configmaps: "20"
    secrets: "20"
    resourcequota.openshift.io requests.storage: 100Gi
EOF

oc apply -f resourcequota.yaml -n my-project

# View quota usage
oc adm quota -n my-project
oc describe quota/project-quota -n my-project

# View quota usage as JSON
oc adm quota -n my-project --showusage

Configuring Project Quotas

# Create a project with resource quota
oc new-project my-project --description="My Project"

# Apply ResourceQuota to the project
cat > project-quota.yaml <<'EOF'
apiVersion: v1
kind: ResourceQuota
metadata:
  name: compute-resources
spec:
  hard:
    requests.cpu: "2"
    requests.memory: 4Gi
    limits.cpu: "4"
    limits.memory: 8Gi
EOF

oc apply -f project-quota.yaml -n my-project

# Storage quota
cat > storage-quota.yaml <<'EOF'
apiVersion: v1
kind: ResourceQuota
metadata:
  name: storage-resources
spec:
  hard:
    requests.storage: 50Gi
    persistentvolumeclaims: "5"
EOF

oc apply -f storage-quota.yaml -n my-project

# Count-only quota
cat > count-quota.yaml <<'EOF'
apiVersion: v1
kind: ResourceQuota
metadata:
  name: object-counts
spec:
  hard:
    pods: "10"
    services: "5"
    persistentvolumeclaims: "5"
    configmaps: "10"
    secrets: "10"
    resourcequota.openshift.io requests.storage: 25Gi
EOF

oc apply -f count-quota.yaml -n my-project

Configuring Project Resource Requirements

# LimitRange: Enforce minimum/maximum/default resource requests
cat > limitrange.yaml <<'EOF'
apiVersion: v1
kind: LimitRange
metadata:
  name: default-limits
spec:
  limits:
  # Default container limits
  - type: Container
    default:
      memory: "512Mi"
      cpu: "500m"
    defaultRequest:
      memory: "256Mi"
      cpu: "250m"
    max:
      memory: "2Gi"
      cpu: "2"
    min:
      memory: "64Mi"
      cpu: "50m"
  # Default pod limits
  - type: Pod
    max:
      cpu: "4"
      memory: "4Gi"
    min:
      cpu: "50m"
      memory: "64Mi"
  # PVC limits
  - type: PersistentVolumeClaim
    max:
      storage: "10Gi"
    min:
      storage: "1Gi"
EOF

oc apply -f limitrange.yaml -n my-project

# View LimitRange
oc describe limitrange/default-limits -n my-project

Configuring Project Limit Ranges

# LimitRange for different resource types
cat > limitrange-multi.yaml <<'EOF'
apiVersion: v1
kind: LimitRange
metadata:
  name: multi-resource-limits
spec:
  limits:
  # Limits for containers
  - type: Container
    max:
      memory: "4Gi"
      cpu: "4"
    min:
      memory: "32Mi"
      cpu: "10m"
    default:
      memory: "512Mi"
      cpu: "500m"
    defaultRequest:
      memory: "256Mi"
      cpu: "250m"
  # Limits for pods
  - type: Pod
    max:
      memory: "8Gi"
      cpu: "8"
  # Limits for PVCs
  - type: PersistentVolumeClaim
    max:
      storage: "20Gi"
    min:
      storage: "1Gi"
    default:
      storage: "5Gi"
  # Limits for requests
  - type: ResourceClaim
    max:
      memory: "2Gi"
      cpu: "2"
    default:
      memory: "256Mi"
      cpu: "250m"
EOF

oc apply -f limitrange-multi.yaml -n my-project

Configuring Project Templates

# Create a custom project template
cat > project-template.yaml <<'EOF'
apiVersion: template.openshift.io/v1
kind: Template
metadata:
  name: my-app-template
  annotations:
    description: "Template for creating a standard application"
    icon-url: "https://www.example.com/icon.png"
objects:
- apiVersion: apps/v1
  kind: Deployment
  metadata:
    name: ${APP_NAME}
  spec:
    replicas: ${REPLICAS:-1}
    selector:
      matchLabels:
        app: ${APP_NAME}
    template:
      metadata:
        labels:
          app: ${APP_NAME}
      spec:
        containers:
        - name: ${APP_NAME}
          image: ${IMAGE:-nginx:latest}
          ports:
          - containerPort: ${PORT:-80}
          resources:
            requests:
              cpu: ${CPU_REQ:-250m}
              memory: ${MEM_REQ:-256Mi}
            limits:
              cpu: ${CPU_LIM:-500m}
              memory: ${MEM_LIM:-512Mi}
- apiVersion: v1
  kind: Service
  metadata:
    name: ${APP_NAME}-svc
  spec:
    selector:
      app: ${APP_NAME}
    ports:
    - port: 80
      targetPort: ${PORT:-80}
- apiVersion: route.openshift.io/v1
  kind: Route
  metadata:
    name: ${APP_NAME}
  spec:
    to:
      kind: Service
      name: ${APP_NAME}-svc
parameters:
- name: APP_NAME
  required: true
- name: REPLICAS
  value: "1"
  required: false
- name: IMAGE
  value: nginx:latest
  required: false
- name: PORT
  value: "80"
  required: false
- name: CPU_REQ
  value: "250m"
  required: false
- name: MEM_REQ
  value: "256Mi"
  required: false
- name: CPU_LIM
  value: "500m"
  required: false
- name: MEM_LIM
  value: "512Mi"
  required: false
EOF

oc apply -f project-template.yaml -n my-project

# Use the template
oc process -f project-template.yaml -p APP_NAME=myapp -p IMAGE=myregistry.com/myapp:latest | oc apply -f -

# List available templates
oc get templates -n my-project

Quota Summary and Best Practices

# Check all quotas in a project
oc get quota -n my-project

# Check detailed quota usage
oc adm quota -n my-project --showusage

# Check LimitRanges
oc get limitrange -n my-project
oc describe limitrange <limit-name> -n my-project

# Quota hierarchy:
#   1. LimitRange: Sets defaults and constraints for containers/pods/PVCs in a namespace
#   2. ResourceQuota: Sets aggregate resource limits for a namespace
#   3. ClusterAdmin: Sets namespace-level defaults

# Enable default resource requirements in a project
cat > namespace-resources.yaml <<'EOF'
apiVersion: v1
kind: ResourceQuota
metadata:
  name: default-quota
spec:
  hard:
    requests.cpu: "4"
    requests.memory: 8Gi
    limits.cpu: "8"
    limits.memory: 16Gi
    pods: "50"
EOF

# Apply LimitRange for defaults
cat > default-limits.yaml <<'EOF'
apiVersion: v1
kind: LimitRange
metadata:
  name: default-limits
spec:
  limits:
  - type: Container
    default:
      cpu: "500m"
      memory: "512Mi"
    defaultRequest:
      cpu: "250m"
      memory: "256Mi"
EOF

oc apply -f namespace-resources.yaml -n my-project
oc apply -f default-limits.yaml -n my-project
